    <script>

        const video = document.getElementById('camera-view');

        const canvas = document.getElementById('ar-canvas');

        const ctx = canvas.getContext('2d');

        const permissionScreen = document.getElementById('permission-screen');

        const startBtn = document.getElementById('start-ar-btn');

        const modeText = document.getElementById('current-mode');

        const elementBtns = document.querySelectorAll('.element-btn');

        let currentElement = 'pond';

        let ripples = [];

        let particles = [];

        let inkLines = [];

        // Resize Canvas

        function resizeCanvas() {

            canvas.width = window.innerWidth;

            canvas.height = window.innerHeight;

            initInkLines();

        }

        window.addEventListener('resize', resizeCanvas);

        // Initialize organic sacred geometry ink lines across screen

        function initInkLines() {

            inkLines = [];

            const numCenters = 3;

            for (let i = 0; i < numCenters; i++) {

                let cx = Math.random() * canvas.width;

                let cy = Math.random() * canvas.height;

                let rings = 5;

                for (let r = 1; r <= rings; r++) {

                    inkLines.push({

                        cx: cx,

                        cy: cy,

                        radius: r * 40,

                        angleOffset: Math.random() * Math.PI,

                        speed: (Math.random() - 0.5) * 0.005

                    });

                }

            }

        }

        // Start Camera Stream (Back camera preferred for AR room integration)

        async function startCamera() {

            try {

                const constraints = {

                    video: {

                        facingMode: { ideal: 'environment' },

                        width: { ideal: 1920 },

                        height: { ideal: 1080 }

                    },

                    audio: false

                };

                const stream = await navigator.mediaDevices.getUserMedia(constraints);

                video.srcObject = stream;

                permissionScreen.style.display = 'none';

                resizeCanvas();

                requestAnimationFrame(render);

            } catch (err) {

                console.error("Camera access error:", err);

                alert("Camera access is required to blend your real room into the art portal.");

            }

        }

        startBtn.addEventListener('click', startCamera);

        // Handle Touch / Click Interactions

        window.addEventListener('pointerdown', (e) => {

            // Ignore clicks on bottom bar

            if (e.target.closest('.elements-bar') || e.target.closest('.hud-header')) return;

            const rect = canvas.getBoundingClientRect();

            const x = e.clientX - rect.left;

            const y = e.clientY - rect.top;

            // Add water ripple

            ripples.push({ x: x, y: y, radius: 5, maxRadius: 180, alpha: 0.8 });

            // Spawn element particles based on active mode

            spawnElementParticles(x, y, currentElement);

        });

        // Switch Elements

        elementBtns.forEach(btn => {

            btn.addEventListener('click', (e) => {

                elementBtns.forEach(b => b.classList.remove('active'));

                btn.classList.add('active');

                currentElement = btn.getAttribute('data-element');

               

                const names = {

                    pond: "Lotus Pond Stillness",

                    earth: "Prithvi - Roots & Soil Blooming",

                    water: "Jal - Silver Liquid Flow",

                    fire: "Agni - Sacred Amber Glow",

                    air: "Vayu - Wind & Breath Currents",

                    space: "Akasha - Cosmic Void & Portals"

                };

                modeText.textContent = names[currentElement] || "Interactive Portal Active";

            });

        });

        function spawnElementParticles(x, y, type) {

            let count = 25;

            for (let i = 0; i < count; i++) {

                let angle = Math.random() * Math.PI * 2;

                let speed = Math.random() * 3 + 1;

                particles.push({

                    x: x,

                    y: y,

                    vx: Math.cos(angle) * speed,

                    vy: Math.sin(angle) * speed,

                    radius: Math.random() * 4 + 2,

                    alpha: 1.0,

                    life: Math.random() * 60 + 40,

                    type: type

                });

            }

        }

        // Render Loop

        function render() {

            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // 1. Draw Organic/Sacred Geometry Ink Lines blending into room

            ctx.lineWidth = 1.5;

            inkLines.forEach(line => {

                line.angleOffset += line.speed;

                ctx.strokeStyle = getElementInkColor(currentElement, 0.35);

                ctx.beginPath();

                // Draw wavy organic circles

                let points = 60;

                for (let i = 0; i <= points; i++) {

                    let theta = (i / points) * Math.PI * 2;

                    let distortion = Math.sin(theta * 6 + line.angleOffset) * 6;

                    let r = line.radius + distortion;

                    let px = line.cx + Math.cos(theta) * r;

                    let py = line.cy + Math.sin(theta) * r;

                    if (i === 0) ctx.moveTo(px, py);

                    else ctx.lineTo(px, py);

                }

                ctx.closePath();

                ctx.stroke();

            });

            // 2. Render Ripples (Lotus Pond effect)

            for (let i = ripples.length - 1; i >= 0; i--) {

                let rip = ripples[i];

                rip.radius += 2.5;

                rip.alpha *= 0.96;

                ctx.strokeStyle = getElementRippleColor(currentElement, rip.alpha);

                ctx.lineWidth = 2;

                ctx.beginPath();

                ctx.arc(rip.x, rip.y, rip.radius, 0, Math.PI * 2);

                ctx.stroke();

                if (rip.alpha < 0.02 || rip.radius > rip.maxRadius) {

                    ripples.splice(i, 1);

                }

            }

            // 3. Render Element Particles

            for (let i = particles.length - 1; i >= 0; i--) {

                let p = particles[i];

                p.x += p.vx;

                p.y += p.vy;

                p.alpha -= 1 / p.life;

                ctx.fillStyle = getParticleColor(p.type, p.alpha);

                ctx.beginPath();

                ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);

                ctx.fill();

                if (p.alpha <= 0) {

                    particles.splice(i, 1);

                }

            }

            requestAnimationFrame(render);

        }

        function getElementInkColor(type, alpha) {

            switch(type) {

                case 'earth': return `rgba(139, 115, 85, ${alpha})`;

                case 'water': return `rgba(100, 180, 220, ${alpha})`;

                case 'fire': return `rgba(220, 120, 50, ${alpha})`;

                case 'air': return `rgba(200, 220, 200, ${alpha})`;

                case 'space': return `rgba(180, 130, 220, ${alpha})`;

                default: return `rgba(242, 239, 233, ${alpha})`;

            }

        }

        function getElementRippleColor(type, alpha) {

            switch(type) {

                case 'earth': return `rgba(160, 130, 90, ${alpha})`;

                case 'water': return `rgba(255, 255, 255, ${alpha})`;

                case 'fire': return `rgba(255, 160, 60, ${alpha})`;

                case 'air': return `rgba(180, 240, 220, ${alpha})`;

                case 'space': return `rgba(210, 175, 55, ${alpha})`;

                default: return `rgba(255, 255, 255, ${alpha})`;

            }

        }

        function getParticleColor(type, alpha) {

            switch(type) {

                case 'earth': return `rgba(120, 95, 60, ${alpha})`;

                case 'water': return `rgba(130, 210, 245, ${alpha})`;

                case 'fire': return `rgba(255, 100, 30, ${alpha})`;

                case 'air': return `rgba(230, 240, 230, ${alpha})`;

                case 'space': return `rgba(212, 175, 55, ${alpha})`;

                default: return `rgba(242, 239, 233, ${alpha})`;

            }

        }
